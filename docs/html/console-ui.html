# allow_safe_tool — permits read-only and safe-write tools whose
# side-effect class is within the agent's delegated authority.
package aumos.fin

import future.keywords.if

default allow = false

allow if {
    safe_side_effect(input.authority.side_effect_class)
    within_spend_budget(input.authority)
    input.attestation.verified == true
    input.authority.expiry > time.now_ns()
}

safe_side_effect(cls) if {
    cls == "read"
}

safe_side_effect(cls) if {
    cls == "write"
    input.tool.blast_radius == "low"
}

within_spend_budget(auth) if {
    input.action.amount <= auth.spend_budget
}
